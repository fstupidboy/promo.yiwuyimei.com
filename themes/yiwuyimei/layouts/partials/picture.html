{{/* Enhanced component: supports AVIF/WebP fallbacks, responsive sizes, optional fetchpriority. Ensures uses a broadly supported fallback (jpg/png/webp) when original is .avif. */}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $decoding := .decoding | default "async" -}} {{- $class := .class -}} {{- $width := .width -}} {{- $height := .height -}} {{- $sizes := .sizes | default "100vw" -}} {{- $fetchPriority := .fetchpriority | default "auto" -}} {{- $srcNoSlash := replaceRE "^/" "" $src -}} {{- $staticPath := printf "static/%s" $srcNoSlash -}} {{- $ext := path.Ext $staticPath -}} {{- $baseNoExt := replace $staticPath $ext "" -}} {{- $baseURLNoExt := replace $srcNoSlash $ext "" -}} {{- $avifStatic := printf "%s.avif" $baseNoExt -}} {{- $webpStatic := printf "%s.webp" $baseNoExt -}} {{- $jpgStatic := printf "%s.jpg" $baseNoExt -}} {{- $jpegStatic := printf "%s.jpeg" $baseNoExt -}} {{- $pngStatic := printf "%s.png" $baseNoExt -}} {{- $avifURL := printf "/%s.avif" $baseURLNoExt -}} {{- $webpURL := printf "/%s.webp" $baseURLNoExt -}} {{- /* Encode spaces/parentheses for attribute safety */ -}} {{- $avifAttr := ($avifURL | relURL | safeURL) -}} {{- $webpAttr := ($webpURL | relURL | safeURL) -}} {{- $jpgURL := printf "/%s.jpg" $baseURLNoExt -}} {{- $jpegURL := printf "/%s.jpeg" $baseURLNoExt -}} {{- $pngURL := printf "/%s.png" $baseURLNoExt -}} {{- /* Compute a safe fallback for : prefer jpg/jpeg/png, then webp, else original */ -}} {{- $imgFallback := $src -}} {{- if fileExists $jpgStatic -}} {{- $imgFallback = $jpgURL -}} {{- else if fileExists $jpegStatic -}} {{- $imgFallback = $jpegURL -}} {{- else if fileExists $pngStatic -}} {{- $imgFallback = $pngURL -}} {{- else if fileExists $webpStatic -}} {{- $imgFallback = $webpURL -}} {{- end -}} {{- if fileExists $avifStatic }} {{- end }} {{- if fileExists $webpStatic }} {{- end }} {{- $imgAttr := ($imgFallback | relURL | safeURL) -}}
